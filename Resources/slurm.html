SLURM Guide & Cheatsheet

Essential SLURM Commands

Job Management

sbatch [script.sh]
Submit a job script.
squeue
Check job status (all users).
squeue -u $USER
Check your jobs.
scontrol show job [jobid]
View job details.
scancel [jobid]
Cancel a specific job.
scancel -u $USER
Cancel all your jobs.

Resource Information

sinfo
View node status and partitions.
sinfo -o "%P %G %D %N %T"
Check node status (detailed).
sstat --format=JobID,JobName,User,CPUTime,AveRSS
View cluster usage.

Interactive Sessions

salloc --nodes=1 --cpus-per-task=4 --time=01:00:00
Allocate resources for an interactive shell.
srun --pty bash -i
Start an interactive shell on a compute node (requires prior salloc or equivalent).

# SLURM Custom Commands and Aliases


# Alias: Check your running jobs (Assuming 'ymachta' is the username)
alias wyd='squeue -u ymachta'

# Alias: View node information (Partition, GRES, Cores, Node Names)
alias nodes='sinfo -o "%P %G %D %N"'

# Function: Allocate resources for GPU development
# Usage: gpu <GPU_TYPE> [COUNT] [-h HOURS]
# Example: gpu a100 1 -h 8  (Allocates 1 A100 GPU for 8 hours)
gpu() {
    local hours=8
    local gpu_type=""
    local gpu_count=1

    # Parse arguments
    while [[ $# -gt 0 ]]; do
        case "$1" in
            -h|--hours)
                hours="$2"
                shift 2
                ;;
            *)
                # Capture GPU type and count
                if [[ -z "$gpu_type" ]]; then
                    gpu_type="$1"
                elif [[ -z "$gpu_count" && "$1" =~ ^[0-9]+$ ]]; then
                    gpu_count="$1"
                fi
                shift
                ;;
        esac
    done

    # Construct the GRES argument
    if [ -n "$gpu_type" ]; then
        gres_arg="--gres=gpu:${gpu_type}:${gpu_count}"
    else
        # Default to generic GPU if type not specified, ensuring count is used
        gres_arg="--gres=gpu:${gpu_count}"
    fi

    # Display chosen arguments before execution
    echo "Allocating: $gres_arg --time=$(printf '%02d:00:00' "$hours")"

    # Start the interactive session
    salloc --partition=gpu \
           $gres_arg \
           --cpus-per-task=4 \
           --mem=64G \
           --time=$(printf '%02d:00:00' "$hours") \
           --job-name=dev \
           --ntasks=1 \
           --nodes=1
}
